# Бриф для сборки макетов Car Reminder в Pencil

Файл: `/Users/badfish/projects/car-reminder/design.pen`

Первым делом вызови:
`get_app_state({include_schema:true, include_canvas_design:true, include_scripts_and_shaders:false})`
— без схемы .pen работать нельзя.

## Правила совместной работы

- Твой экран — уже созданный фрейм 390×844 (`layout:"vertical"`, `clip:true`, `fill:"$bg"`,
  `placeholder:true`). Вставляй узлы **только внутрь него**.
- Никогда не вставляй в `document` и не трогай чужие узлы — над соседними экранами
  параллельно работают другие агенты.
- Когда экран готов и проверен: `Update(SCREEN,{placeholder:false})`.

## Критичные особенности этой сборки

- **`type:"ref"` не отрисовывается.** Не используй инстансы компонентов и `Copy` от
  reusable-узлов. Всё строится обычными `frame`/`text`/`icon`/`ellipse`, а повторы —
  JS-циклами и хелпер-функциями внутри одного `execute`.
- **Глобальные переменные между вызовами `execute` не сохраняются.** Используй строковые id
  из ответа предыдущего вызова.
- У `text` всегда указывай `fill`, иначе он невидим.
- Перенос строк — только `textGrowth:"fixed-width"` + `width:"fill_container"`.
- Нельзя: проценты в размерах, `margin`, `alignItems:"baseline"/"stretch"`.
- `x`/`y` игнорируются внутри flex-родителя.
- У каждого узла обязателен `name`.

## Токены (уже заведены, ссылайся через `$имя`)

Цвета: `$bg` `$surface` `$surface-alt` `$border` `$text-primary` `$text-secondary`
`$text-tertiary` `$accent` `$accent-soft` `$ok` `$warn` `$danger` `$ring-track`
`$ring-track-danger` `$muted`

Шрифты: `$font-ui` (Inter) — весь текст; `$font-mono` (JetBrains Mono) — **только цифры**:
пробег, километры, суммы, даты внутри списков.

Скругления: карточки 16–20, поля и мелкие блоки 10–14. Тени не используем.

## Рецепты

Статус-бар — всегда первый ребёнок экрана:

```js
sb=Insert(SCREEN,{type:"frame",name:"Status Bar",width:"fill_container",height:46,layout:"horizontal",justifyContent:"space_between",alignItems:"center",padding:[0,22]})
Insert(sb,{type:"text",name:"Time",content:"9:41",fontFamily:"$font-ui",fontSize:14,fontWeight:"600",fill:"$text-primary"})
ind=Insert(sb,{type:"frame",name:"Indicators",layout:"horizontal",gap:6,alignItems:"center"})
for(const i of ["signal-high","wifi","battery-full"])Insert(ind,{type:"icon",name:i,library:"lucide",icon:i,width:15,height:15,fill:"$text-primary"})
```

Шапка с кнопкой «назад»:

```js
ab=Insert(SCREEN,{type:"frame",name:"App Bar",width:"fill_container",height:54,layout:"horizontal",gap:2,alignItems:"center",padding:[0,10]})
bk=Insert(ab,{type:"frame",name:"Back",width:38,height:38,justifyContent:"center",alignItems:"center"})
Insert(bk,{type:"icon",name:"Back Icon",library:"lucide",icon:"arrow-left",width:21,height:21,fill:"$text-primary"})
Insert(ab,{type:"text",name:"Title",content:"Заголовок",fontFamily:"$font-ui",fontSize:17,fontWeight:"600",fill:"$text-primary",textGrowth:"fixed-width",width:"fill_container"})
```

Контентная область между шапкой и навигацией:

```js
content=Insert(SCREEN,{type:"frame",name:"Content",width:"fill_container",height:"fill_container",layout:"vertical",gap:16,padding:[8,16,16,16]})
```

Заголовок секции:

```js
Insert(content,{type:"text",name:"Section Label",content:"ТРЕБУЕТ ВНИМАНИЯ",fontFamily:"$font-ui",fontSize:11,fontWeight:"600",letterSpacing:0.8,fill:"$text-tertiary"})
```

Бублик. `value` 0–100, где 100 — новая деталь, 0 — пора менять:

```js
function donut(parent,size,value,color,track,label){
  const d=Insert(parent,{type:"frame",name:"Donut",layout:"none",width:size,height:size})
  Insert(d,{type:"ellipse",name:"Track",x:0,y:0,width:size,height:size,innerRadius:size>100?0.82:0.76,fill:track})
  Insert(d,{type:"ellipse",name:"Arc",x:0,y:0,width:size,height:size,innerRadius:size>100?0.82:0.76,fill:color,startAngle:90,sweepAngle:-(value/100)*360})
  const w=Insert(d,{type:"frame",name:"Label Wrap",x:0,y:0,width:size,height:size,justifyContent:"center",alignItems:"center"})
  Insert(w,{type:"text",name:"Value",content:label,fontFamily:"$font-mono",fontSize:Math.round(size*0.3),fontWeight:"600",fill:color})
  return d
}
```

Состояния бублика:
- норма — `color:"$ok"`, `track:"$ring-track"`;
- скоро — `color:"$warn"`, `track:"$ring-track"`;
- просрочено — `value:0`, `color:"$danger"`, `track:"$ring-track-danger"`, `label:"0"`;
- нет данных — `value:0`, `color:"$text-tertiary"`, `track:"$ring-track"`, `label:"—"`.

function field(parent,label,value,suffix,muted){
  const f=Insert(parent,{type:"frame",name:"Field "+label,width:"fill_container",layout:"vertical",gap:7})

## Тон дизайна

- Спокойный, утилитарный, без декора: цвет несёт смысл (зелёный/жёлтый/красный/серый —
  состояние детали), синий — только действия.
- Не заворачивай каждый элемент в свою карточку. Карточка нужна там, где это реальная
  группа.
- Цифры — моноширинным, это даёт приборный характер. Подписи — Inter.

## Проверка перед снятием placeholder

**Два ложных сигнала в этом документе — не верь им вслепую:**

- `ctx.bounds.y` у детей завышен примерно на 50 px на любой глубине, поэтому `c.problems`
  массово рапортует `partially/fully clipped` даже на заведомо корректной вёрстке. Не
  сокращай контент только из-за этого — посчитай суммарную высоту детей вручную
  (46 статус-бар + 54 шапка + контент + 70 навигация ≤ 844).
- `get_screenshot` кеширует картинку по id узла: если узел снимался, когда был пустым,
  дальше отдаётся та же пустая картинка.

1. Проверь фит арифметикой по высотам детей, а не только по `c.problems`.
2. Посмотри глазами через свежую копию (кеш скриншота обходится новым id):
   `probe=Copy(SCREEN,document,{name:"ZZ Probe",x:4200,y:560})` → `get_screenshot` по `probe`
   → `Delete(probe)`. Ничего не обрезано, текст читаем, бублики круглые, колонки выровнены.
3. Правь `Update`'ами, не пересоздавай экран заново.
